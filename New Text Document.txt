# cert_utils.py
from cryptography import x509
from cryptography.hazmat.primitives import hashes, serialization
from cryptography.hazmat.primitives.asymmetric import padding
from cryptography.x509.oid import NameOID, ExtensionOID
from datetime import datetime, timezone

class CertValidationError(Exception):
    pass

def load_pem_cert(pem_bytes: bytes) -> x509.Certificate:
    return x509.load_pem_x509_certificate(pem_bytes)

def compute_cert_sha256_fingerprint(cert: x509.Certificate) -> str:
    # returns hex string of SHA-256 over DER bytes
    der = cert.public_bytes(serialization.Encoding.DER)
    digest = hashes.Hash(hashes.SHA256())
    digest.update(der)
    return digest.finalize().hex()

def validate_cert_signed_by_ca(cert: x509.Certificate, ca_cert: x509.Certificate) -> None:
    """
    Verifies the signature on 'cert' using ca_cert's public key.
    Raises CertValidationError on failure.
    """
    pubkey = ca_cert.public_key()
    try:
        pubkey.verify(
            signature=cert.signature,
            data=cert.tbs_certificate_bytes,
            padding=padding.PKCS1v15(),  # typical for RSA-signed certs; adapt if using other algs
            algorithm=cert.signature_hash_algorithm,
        )
    except Exception as e:
        raise CertValidationError(f"Signature verification failed: {e}")

def validate_cert_dates(cert: x509.Certificate) -> None:
    now = datetime.now(timezone.utc)
    if cert.not_valid_before > now:
        raise CertValidationError(f"Certificate not yet valid: not_valid_before={cert.not_valid_before}")
    if cert.not_valid_after < now:
        raise CertValidationError(f"Certificate expired: not_valid_after={cert.not_valid_after}")

def get_common_name(cert: x509.Certificate) -> str:
    try:
        cn_attr = cert.subject.get_attributes_for_oid(NameOID.COMMON_NAME)
        return cn_attr[0].value if cn_attr else ""
    except Exception:
        return ""

def validate_san_if_present(cert: x509.Certificate, expected_names: list[str]) -> None:
    """
    If SAN extension exists and expected_names provided, ensure at least one SAN matches.
    Not mandatory for CN-only setups; optional extra check.
    """
    try:
        san = cert.extensions.get_extension_for_oid(ExtensionOID.SUBJECT_ALTERNATIVE_NAME).value
    except x509.ExtensionNotFound:
        return  # no SAN -> skip
    san_names = []
    san_names.extend(san.get_values_for_type(x509.DNSName))
    # compare any match
    for exp in expected_names:
        if exp in san_names:
            return
    raise CertValidationError(f"No matching SAN found. SANs={san_names} expected one of {expected_names}")

def validate_peer_certificate(peer_pem: bytes, ca_pem_path: str, expected_cn: str, expected_san_names: list[str]|None=None) -> dict:
    """
    Main convenience function.
    Returns dict with metadata on success.
    Raises CertValidationError on any validation failure.
    """
    # load certs
    peer_cert = load_pem_cert(peer_pem)
    with open(ca_pem_path, "rb") as f:
        ca_cert = load_pem_cert(f.read())

    # 1. signature chain check (CA -> peer)
    validate_cert_signed_by_ca(peer_cert, ca_cert)

    # 2. validity dates
    validate_cert_dates(peer_cert)

    # 3. CN check
    cn = get_common_name(peer_cert)
    if cn != expected_cn:
        raise CertValidationError(f"CN mismatch: expected='{expected_cn}', got='{cn}'")

    # 4. optional SAN check
    if expected_san_names:
        validate_san_if_present(peer_cert, expected_san_names)

    # metadata to use (fingerprint etc.)
    fp = compute_cert_sha256_fingerprint(peer_cert)
    return {
        "subject_cn": cn,
        "not_valid_before": peer_cert.not_valid_before,
        "not_valid_after": peer_cert.not_valid_after,
        "fingerprint_sha256": fp,
        "issuer": peer_cert.issuer.rfc4514_string()
    }
